# ----------- Day-09 : Docker Integration with Jenkins and Gitlab -----------
# 1. Ansible - Setup Docker for Nginx

---
- name: Setup Docker container for Nginx
  hosts: all
  become: yes

  tasks:
    - name: Pull Nginx Docker image
        community.docker.docker_image:
          name: nginx
          source: pull

    - name: Run Nginx Container
        community.docker.docker_container:
           name: nginx-container
           image: nginx
           state: started
           ports:
              - "8080:80"


# 2. Terraform - EC2 with auto scaling group

resource "aws_launch_configuration" "example"{
    name = "example-config"
    image_id = "0abcd1234abcd1234"
    instance_type = "t3.medium"    
}

resource "aws_autoscaling_group" "example"{
  name = "example-asg"
  desired_capacity = 2
  max_size = 3
  min_size = 1
  vpc_zone_identifier = "[subnet-12345678]"
  launch_configuration = aws_launch_configuration.example.id
}


# 3. Kubernetes - Horizontal pod autoscaler
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      labels: 
        app: pathnex-app

    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
             - containerPort: 80
---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: pathnex-deployment

  minReplicas: 1       
  maxReplicas: 10

  metrics:
    - type: resource
      resource:
        name: cpu
        target:
          type: Utilization
          averageUtilization: 50


# 4. Jenkins - Deploy to Kubernetes

pipeline{
  agent any

  stages{
    stage('Deploy to Kubernetes'){
      steps{
        sh 'kubectl apply -f deployment.yaml'
      }
    }
  }
}


# 5. Gitlab CI/CD - Deploy to Kubernetes

stages:
  - deploy

deploy:
  stage: deploy
  script:
    - kubectl apply -f deployment.yaml
